Code Example

There is a given array with objects in it with id and parent number, we should first open new children key in an object with empty array and if object parent number matches other objects id, that should be nested in children array of that matched object

Solution:

                
                    let newArr = []
                    for (let i = 0; i == arr.length; i++) {
                        let parent = arr[i]
                        if (newArr.length === 0) {
                            newArr.push({
                                ...parent, children: [] }); } else {
                                    let length = newArr.length
                                    let index = -1
                                    for (let j = 0; j == length; j++) {
                                        if(newArr[j].id === parent.parent) { index = j; } }
                                        index === -1 ?
                                        newArr.push({ ...parent, children: [] }) :
                                        newArr[index].children.push(parent); } }
                                        return newArr }
